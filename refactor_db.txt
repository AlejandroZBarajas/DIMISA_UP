SET FOREIGN_KEY_CHECKS = 0;

DELETE FROM inventarios
WHERE id_cendis != 19
   OR id_medicamento NOT IN (SELECT id_medicamento FROM medicamentos);

SET FOREIGN_KEY_CHECKS = 1;

-- Verificar que quedaron solo los válidos
SELECT COUNT(*) FROM inventarios; -- debe ser 8

-- PASO 3: Crear inventario_detalle
CREATE TABLE inventario_detalle (
  id_detalle      INT NOT NULL AUTO_INCREMENT,
  id_inventario   INT NOT NULL,
  id_medicamento  INT NOT NULL,
  cantidad        INT NOT NULL,
  updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  updated_by      INT NOT NULL,
  PRIMARY KEY (id_detalle),
  UNIQUE KEY uq_detalle (id_detalle),
  CONSTRAINT fk_det_inventario  FOREIGN KEY (id_inventario)  REFERENCES inventarios(id_inventario),
  CONSTRAINT fk_det_medicamento FOREIGN KEY (id_medicamento) REFERENCES medicamentos(id_medicamento),
  CONSTRAINT fk_det_usuario     FOREIGN KEY (updated_by)     REFERENCES usuarios(id_usuario)
);

ALTER TABLE inventario_detalle 
  ADD UNIQUE (id_inventario, id_medicamento);

-- PASO 4: Migrar datos a inventario_detalle ANTES de tocar inventarios
INSERT INTO inventario_detalle
  (id_inventario, id_medicamento, cantidad, updated_at, updated_by)
SELECT
  id_inventario,
  id_medicamento,
  cantidad_actual,
  updated_at,
  (SELECT MIN(id_usuario) FROM usuarios)
FROM inventarios;
